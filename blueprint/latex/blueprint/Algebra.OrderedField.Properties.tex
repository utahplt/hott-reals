\section{\texttt{Algebra.OrderedField.Properties}}


\begin{lemma}[\agda{·CancelR<}]
  \label{mulCancelRLt}
  Let $F$ be an ordered field (\Cref{IsOrderedField}). Multiplication by a positive element reflects strict inequality. For all $x, y, z : F$, if $0 < z$ and $x \cdot z < y \cdot z$, then $x < y$.
\end{lemma}
