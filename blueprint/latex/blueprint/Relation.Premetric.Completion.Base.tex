\section{\texttt{Relation.Premetric.Completion.Base}}

\begin{definition}[\agda{ℭ}, \agda{\_∼[\_]\_}; {\textcite[Definition 3.1]{gilbert2017}}]
  \label{completion}
  The \textbf{completion} of a premetric space $M$ is a type $\completion{M}$ together with a family of mere relations $u \cclose{\varepsilon} v$ on $\completion{M}$, indexed by $\varepsilon : \Qp$, defined simultaneously as a higher inductive-inductive type family. The type $\completion{M}$ comes equipped with constructors:
  \begin{itemize}
  \item a map $\iota \colon M \to \completion{M}$;
  \item for every Cauchy approximation $x$ in $\completion{M}$ with respect to $\cclose{\varepsilon}$ (\Cref{isCauchy}), an element $\lim x : \completion{M}$;
  \item (separatedness) if $u \cclose{\varepsilon} v$ for every $\varepsilon$, then $u = v$.
  \end{itemize}
  Together, the relation $u \cclose{\varepsilon} v$ on $\completion{M}$ is generated by the constructors:
  \begin{itemize}
  \item if $a \close{\varepsilon} b$ in $M$, then $\iota(a) \cclose{\varepsilon} \iota(b)$;
  \item if $\iota(a) \cclose{\varepsilon} y_\delta$, then $\iota(a) \cclose{\varepsilon + \delta} \lim y$;
  \item if $x_\delta \cclose{\varepsilon} \iota(b)$, then $\lim x \cclose{\varepsilon + \delta} \iota(b)$;
  \item if $x_\delta \cclose{\varepsilon} y_\eta$, then $\lim x \cclose{\varepsilon + \delta + \eta} \lim y$;
  \item if $\varphi, \varphi' : u \cclose{\varepsilon} v$, then $\varphi = \varphi'$.
  \end{itemize}
  The last constructor says that each $u \cclose{\varepsilon} v$ is a proposition.
\end{definition}

\textcite[Definition 3.1]{gilbert2017} and \textcite[Definition 11.3.2]{univalentfoundationsprogram2013} state the second, third and fourth constructors of $\cclose{\varepsilon}$ with the tolerance subtracted from the conclusion instead of added to the hypothesis: e.g. for $\delta < \varepsilon$, $\iota(a) \cclose{\varepsilon - \delta} y_\delta$ implies $\iota(a) \cclose{\varepsilon} \lim y$. The two presentations generate the same relation.
