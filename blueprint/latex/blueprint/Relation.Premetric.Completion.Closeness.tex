\begin{lemma}[\agda{∼≃B}; {\textcite[Theorem 3.15]{gilbert2017}}]
  \label{closeIffBall}
  Let $M$ be a premetric space. For all $a, b : M$ and $\varepsilon : \Qp$, $\iota(a) \cclose{\varepsilon} \iota(b)$ if and only if $a \close{\varepsilon} b$.
\end{lemma}
